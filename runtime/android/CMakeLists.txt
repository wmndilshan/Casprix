# CMakeLists.txt ??? Android NDK build for ND/Casprix native app
#
# Builds libMainActivity.so: the Skia-native Android app with no Java.
#
# Cross-compiles:
#   - Casprix runtime (scene graph, layout, widgets, android layers)
#   - Links against pre-built Skia ARM64 static libs
#   - Links EGL + GLES from Android NDK sysroot
#
# Usage:
#   cmake -B build-android \
#         -DCMAKE_TOOLCHAIN_FILE=$NDK/build/cmake/android.toolchain.cmake \
#         -DANDROID_ABI=arm64-v8a \
#         -DANDROID_PLATFORM=android-24 \
#         -DSKIA_DIR=<path/to/skia>
#   cmake --build build-android

cmake_minimum_required(VERSION 3.22)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
project(CasprixAndroid CXX C)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# ?????? Skia paths ????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????
# SKIA_DIR defaults to the Skia source tree used by the repo build helper.
if(NOT DEFINED SKIA_DIR)
    set(SKIA_DIR "${CMAKE_SOURCE_DIR}/../../third_party/skia-src")
endif()

# Pre-built Skia static lib for android-arm64. Prefer an explicit override,
# otherwise look in the source tree output and the legacy copy location.
if(NOT DEFINED SKIA_LIB_DIR)
    if(EXISTS "${SKIA_DIR}/lib-android-arm64/libskia.a")
        set(SKIA_LIB_DIR "${SKIA_DIR}/lib-android-arm64")
    elseif(EXISTS "${SKIA_DIR}/out/casprix-android-arm64/libskia.a")
        set(SKIA_LIB_DIR "${SKIA_DIR}/out/casprix-android-arm64")
    else()
        set(SKIA_LIB_DIR "${PROJECT_ROOT}/third_party/skia/lib-android-arm64")
    endif()
endif()

if(NOT EXISTS "${SKIA_LIB_DIR}/libskia.a")
    message(WARNING "Skia ARM64 lib not found at ${SKIA_LIB_DIR}/libskia.a")
    message(WARNING "Run tools/android/build_skia_arm64.bat or download_skia_arm64.bat first")
endif()

# ?????? Project root (for runtime/skia sources) ?????????????????????????????????????????????????????????????????????????????????????????????????????????
set(PROJECT_ROOT "${CMAKE_SOURCE_DIR}/../..")
set(CPX_ACTIVITY_NAME "MainActivity" CACHE STRING "Output shared library name for NativeActivity")

# ?????? Source files ??????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????
set(CASPRIX_SOURCES
    # Android native entry + EGL + renderer
    ${CMAKE_SOURCE_DIR}/android_main.c
    ${CMAKE_SOURCE_DIR}/android_egl.c
    ${CMAKE_SOURCE_DIR}/android_renderer.cpp
    ${CMAKE_SOURCE_DIR}/android_runtime.c

    ${PROJECT_ROOT}/runtime/skia/skia_c.cpp
    # Core Skia scene graph / layout / widget runtime
    ${PROJECT_ROOT}/runtime/skia/scene_graph.c
    ${PROJECT_ROOT}/runtime/skia/layout.c
    ${PROJECT_ROOT}/runtime/skia/events.c
    ${PROJECT_ROOT}/runtime/skia/widgets.c
    ${PROJECT_ROOT}/runtime/skia/style.c
    ${PROJECT_ROOT}/runtime/skia/text.c
    ${PROJECT_ROOT}/runtime/skia/animation.c
    ${PROJECT_ROOT}/runtime/skia/frame_loop_android.c
)

# User app (generated from .cpx ??? C by the Casprix compiler)
if(DEFINED CPX_APP_SOURCE AND EXISTS "${CPX_APP_SOURCE}")
    list(APPEND CASPRIX_SOURCES "${CPX_APP_SOURCE}")
    message(STATUS "Including user app: ${CPX_APP_SOURCE}")
endif()

# ?????? Shared library: libMainActivity.so ????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????
add_library(MainActivity SHARED ${CASPRIX_SOURCES})

target_include_directories(MainActivity PRIVATE
    ${CMAKE_SOURCE_DIR}                   # android_egl.h, android_runtime.h
    ${PROJECT_ROOT}/runtime/skia          # scene_graph.h, layout.h etc.
    ${SKIA_DIR}                           # top-level (some includes do #include "src/...")
    ${SKIA_DIR}/include                   # public Skia headers
    ${SKIA_DIR}/include/core
    ${SKIA_DIR}/include/gpu
    ${SKIA_DIR}/include/gpu/gl
)

target_compile_definitions(MainActivity PRIVATE
    SK_GL=1
    SK_GANESH=1
    SK_ASSUME_GL_ES=1
)

# ?????? Link libraries ????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????
# Pre-built Skia
if(EXISTS "${SKIA_LIB_DIR}/libskia.a")
    target_link_libraries(MainActivity PRIVATE "${SKIA_LIB_DIR}/libskia.a")
endif()

# Android system libraries (provided by NDK sysroot)
find_library(log-lib    log)
find_library(android-lib android)
find_library(egl-lib    EGL)
find_library(gles-lib   GLESv3)

target_link_libraries(MainActivity PRIVATE
    ${log-lib}
    ${android-lib}
    ${egl-lib}
    ${gles-lib}
    atomic
    m
    z  # zlib ??? always available on Android
)

# ?????? Output settings ?????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????????
set_target_properties(MainActivity PROPERTIES
    OUTPUT_NAME "${CPX_ACTIVITY_NAME}"
)
if(DEFINED CPX_OUTPUT_DIR)
    set_target_properties(MainActivity PROPERTIES
        LIBRARY_OUTPUT_DIRECTORY "${CPX_OUTPUT_DIR}"
        LIBRARY_OUTPUT_DIRECTORY_RELEASE "${CPX_OUTPUT_DIR}"
    )
endif()

if(CMAKE_BUILD_TYPE STREQUAL "Release")
    target_compile_options(MainActivity PRIVATE -O2 -DNDEBUG)
endif()
